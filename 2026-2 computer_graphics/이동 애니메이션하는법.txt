float Timer() {

	double now = glfwGetTime();
	float dt = static_cast<float>(now - lastTime);
	lastTime = now;

	return dt;
}

void Move(float dt) {
	if (isMove_1_shape) {
		float dx = shapes[0].speedX * dt;
		float dy = shapes[0].speedY * dt;
		shapes[0].move(dx, dy);

	}
}
